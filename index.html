// ============================== // 1. گرفتن عناصر HTML // ============================== const searchinput = document.getElementById("search"); const brandFilter = document.getElementById("brand-filter"); const noResults = document.getElementById("no-results"); const allbike = document.querySelectorAll(".product"); // ============================== // 2. تابع جست‌وجو و فیلتر محصولات // ============================== function filterBikes() { // متن جست‌وجو const searchtext = searchinput.value .toLowerCase() .trim(); // برند انتخاب‌شده const selectedBrand = brandFilter.value; // آیا محصولی پیدا شده است؟ let hasResults = false; // بررسی تمام محصولات allbike.forEach(function (bike) { // متن کامل محصول const bikeText = bike.textContent.toLowerCase(); // برند محصول const bikeBrand = bike.dataset.brand; // بررسی تطابق با جست‌وجو const matchesSearch = bikeText.includes(searchtext); // بررسی تطابق با برند const matchesBrand = selectedBrand === "all" || bikeBrand === selectedBrand; // هر دو شرط باید برقرار باشند const shouldShow = matchesSearch && matchesBrand; // نمایش یا مخفی‌کردن محصول bike.style.display = shouldShow ? "" : "none"; // ثبت پیدا شدن محصول if (shouldShow) { hasResults = true; } }); // نمایش پیام در صورت نبودن محصول noResults.hidden = hasResults; } // ============================== // 3. فعال‌کردن جست‌وجوی زنده // ============================== searchinput.addEventListener("input", filterBikes); // ============================== // 4. فعال‌کردن فیلتر برند // ============================== brandFilter.addEventListener("change", filterBikes);
