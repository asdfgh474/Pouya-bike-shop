const searchinput = document.getElementById("search"); const searchbutton = document.getElementById("search-btn"); const allBikes = document.querySelectorAll(".product"); searchbutton.addEventListener("click", function() { const searchtext = searchinput.value.toLowerCase().trim(); allBikes.forEach(function(tBike) { const biketext = tBike.textContent.toLowerCase(); if (searchtext === "" || biketext.includes(searchtext)) { tBike.style.display = ""; } else { tBike.style.display = "none"; } }); });
