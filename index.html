// 1. پیدا کردن عناصر HTML const searchinput = document.getElementById("search"); const searchbutton = document.getElementById("search-btn"); const searchresult = document.getElementById("search-result"); const noResults = document.getElementById("no-results"); const divmahsool = document.getElementById("mas"); const fghmahsool = document.getElementById("hgf"); const brandFilter = document.getElementById("brand-filter"); // 2. پیدا کردن همه کارت‌های محصولات const allbike = document.querySelectorAll(".product"); // 3. تابع جست‌وجو و فیلتر برند function filterBikes() { const searchtext = searchinput.value.toLowerCase().trim(); const selectedBrand = brandFilter.value; // بررسی تک‌تک محصولات allbike.forEach(function(tBike) { const biketext = tBike.textContent.toLowerCase(); // آیا محصول با عبارت جست‌وجو مطابقت دارد؟ const matchesSearch = biketext.includes(searchtext); // آیا محصول با برند انتخاب‌شده مطابقت دارد؟ const matchesBrand = selectedBrand === "all" || biketext.includes(selectedBrand); // نمایش یا مخفی‌کردن محصول tBike.style.display = matchesSearch && matchesBrand ? "" : "none"; }); // بررسی اینکه آیا محصولی باقی مانده است const hasresults = [...allbike].some(function(tBike) { return tBike.style.display !== "none"; }); // نمایش پیام در صورت نبودن نتیجه noResults.hidden = hasresults; // مخفی‌کردن عناصر موردنظر در صورت نبودن نتیجه divmahsool.hidden = !hasresults; fghmahsool.hidden = !hasresults; } // 4. جست‌وجوی زنده هنگام تایپ searchinput.addEventListener("input", filterBikes); // 5. فیلتر هنگام تغییر برند brandFilter.addEventListener("change", filterBikes);
